async function loadPosts() { const res = await fetch("/get_chain"); if (res.ok) { const data = await res.json(); console.log(data); // Debugging line to check the fetched posts const posts = data.chain; posts.reverse().forEach(post => { const postHTML = `
${post.username || 'Anonymous'} added post: ${post.ipfs_cid || 'No Media'} (Block #${post.block_index})

${post.text || ''}

${renderMedia(post.ipfs_cid)}
`; document.getElementById("posts-container").innerHTML += postHTML; }); } else { console.error("Failed to load posts"); } }
